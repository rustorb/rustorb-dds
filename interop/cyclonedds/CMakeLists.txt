cmake_minimum_required(VERSION 3.16)
project(rustorb_interop_cyclonedds C)
find_package(CycloneDDS REQUIRED)
idlc_generate(TARGET hello_type FILES ../idl/HelloWorld.idl)
add_executable(hello hello.c)
target_link_libraries(hello hello_type CycloneDDS::ddsc)
idlc_generate(TARGET keyed_type FILES ../idl/Keyed.idl)
add_executable(keyed keyed.c)
target_link_libraries(keyed keyed_type CycloneDDS::ddsc)
add_executable(liveliness liveliness.c)
target_link_libraries(liveliness hello_type CycloneDDS::ddsc)
idlc_generate(TARGET evolve_v1_type FILES ../idl/EvolveV1.idl)
add_executable(evolve_v1 evolve.c)
target_link_libraries(evolve_v1 evolve_v1_type CycloneDDS::ddsc)
idlc_generate(TARGET evolve_v2_type FILES ../idl/EvolveV2.idl)
add_executable(evolve_v2 evolve.c)
target_compile_definitions(evolve_v2 PRIVATE EVOLVE_V2)
target_link_libraries(evolve_v2 evolve_v2_type CycloneDDS::ddsc)
